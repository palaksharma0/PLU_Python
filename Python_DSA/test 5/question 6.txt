'''
# 6. Food Delivery Route Optimizer
### Problem Statement
A food delivery company stores restaurants and delivery locations.
Database tables:
Restaurant
Delivery
Orders
### Requirements
1. Fetch all pending orders.
2. Build a **Graph** representing restaurants and delivery areas.
3. Find the shortest delivery path using **BFS**.
4. Display delivery order sequence.
5. Mark completed deliveries in SQLite.
### Concepts
* Graph
* BFS
* SQL JOIN
* UPDATE
'''

CREATE DATABASE food_delivery_db;

USE food_delivery_db;

CREATE TABLE restaurant (
    restaurant_id INT PRIMARY KEY,
    restaurant_name VARCHAR(100),
    area VARCHAR(50)
);

CREATE TABLE delivery (
    delivery_id INT PRIMARY KEY,
    customer_name VARCHAR(100),
    delivery_area VARCHAR(50)
);

CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    restaurant_id INT,
    delivery_id INT,
    order_status VARCHAR(20),
    FOREIGN KEY (restaurant_id) REFERENCES restaurant(restaurant_id),
    FOREIGN KEY (delivery_id) REFERENCES delivery(delivery_id)
);

INSERT INTO restaurant VALUES

(3,103,203,'Completed'),
(4,104,204,'Pending');

SELECT * FROM orders
WHERE order_status='Pending';

SELECT
o.order_id,
r.restaurant_name,
d.customer_name,
d.delivery_area,
o.order_status
FROM orders o
JOIN restaurant r
ON o.restaurant_id=r.restaurant_id
JOIN delivery d
ON o.delivery_id=d.delivery_id;

SELECT
o.order_id,
r.restaurant_name,
d.customer_name,
d.delivery_area
FROM orders o
JOIN restaurant r
ON o.restaurant_id=r.restaurant_id
JOIN delivery d
ON o.delivery_id=d.delivery_id
WHERE o.order_status='Pending'
ORDER BY o.order_id;

UPDATE orders
SET order_status='Completed'
WHERE order_id=2;

SELECT * FROM orders;

SELECT * FROM orders
WHERE order_status='Completed';
